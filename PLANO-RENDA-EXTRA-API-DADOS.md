## Sumário Executivo

Este documento apresenta um plano completo para criar uma **API de dados** como fonte de renda extra automatizável. A API fornece dados estruturados em tempo real para desenvolvedores de bots, empresas de marketing e IAs.

**Por que esta opção?**
- Menor trabalho contínuo (criar 1x, vender infinitamente)
- Maior margem de lucro (custo operacional baixíssimo)
- Mercado em crescimento explosivo (IAs precisam de dados)
- Recorrência natural (clientes pagam mensal)
- Tempo de retorno mais rápido (1-2 semanas para primeiro R$)

**Retorno Esperado:**
- Mês 1: R$ 1.000-3.000
- Mês 3: R$ 5.000-12.000
- Mês 6: R$ 15.000-30.000
- Mês 12: R$ 30.000-60.000

**Investimento Inicial:** R$ 40-100
**Investimento Mensal:** R$ 75

```sql
    id SERIAL PRIMARY KEY,
```

### 4.2 Scraper Steam (Exemplo Completo)

```typescript
// src/scrapers/steam-scraper.ts

import puppeteer from 'puppeteer';
import { logger } from '../utils/logger';
import { db } from '../database';

interface SteamGame {
  name: string;
  slug: string;
  currentPrice: number;
  originalPrice: number;
  discountPercent: number;
  url: string;
  imageUrl: string;
}

export class SteamScraper {
  private browser: puppeteer.Browser | null = null;
  private page: puppeteer.Page | null = null;

  async initialize(): Promise<void> {
    this.browser = await puppeteer.launch({
      headless: 'new',
      args: [
        '--no-sandbox',
        '--disable-setuid-sandbox',
        '--disable-dev-shm-usage',
        '--disable-gpu',
      ],
    });
    this.page = await this.browser.newPage();
    
    // Configurar User-Agent realista
    await this.page.setUserAgent(
      'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/125.0.0.0 Safari/537.36'
    );
    
    // Configurar viewport
    await this.page.setViewport({ width: 1920, height: 1080 });
    
    // Bloquear recursos desnecessários
    await this.page.setRequestInterception(true);
    this.page.on('request', (req) => {
      const resourceType = req.resourceType();
      if (['image', 'stylesheet', 'font', 'media'].includes(resourceType)) {
        req.abort();
      } else {
        req.continue();
      }
    });
  }

  async scrapeTopDeals(): Promise<SteamGame[]> {
    if (!this.page) throw new Error('Scraper not initialized');

    const games: SteamGame[] = [];

    try {
      // Navegar para página de ofertas
      await this.page.goto('https://store.steampowered.com/search/?specials=1&ndl=1', {
        waitUntil: 'networkidle2',
        timeout: 30000,
      });

      // Aguardar resultados carregarem
      await this.page.waitForSelector('#search_result_rows', { timeout: 10000 });

      // Extrair dados dos jogos
      const gamesData = await this.page.evaluate(() => {
        const rows = document.querySelectorAll('#search_result_rows a');
        const results: any[] = [];

        rows.forEach((row) => {
          try {
            const nameElement = row.querySelector('.title');
            const priceElement = row.querySelector('.discount_final_price');
            const originalPriceElement = row.querySelector('.discount_original_price');
            const discountElement = row.querySelector('.discount_pct');
            const imageElement = row.querySelector('img');

            if (nameElement && priceElement) {
              const name = nameElement.textContent?.trim() || '';
              const priceText = priceElement.textContent?.trim() || '0';
              const originalPriceText = originalPriceElement?.textContent?.trim() || '0';
              const discountText = discountElement?.textContent?.trim() || '0%';

              // Extrair URL
              const url = row.getAttribute('href') || '';
              const slugMatch = url.match(/\/app\/(\d+)/);
              const slug = slugMatch ? slugMatch[1] : '';

              // Extrair imagem
              const imageUrl = imageElement?.getAttribute('src') || '';

              // Converter preços
              const currentPrice = parseFloat(priceText.replace(/[^\d,]/g, '').replace(',', '.'));
              const originalPrice = parseFloat(originalPriceText.replace(/[^\d,]/g, '').replace(',', '.'));
              const discountPercent = parseInt(discountText.replace(/[^0-9]/g, ''));

              results.push({
                name,
                slug,
                currentPrice,
                originalPrice,
                discountPercent,
                url: `https://store.steampowered.com/app/${slug}`,
                imageUrl,
              });
            }
          } catch (e) {
            // Ignorar erros em linhas individuais
          }
        });

        return results;
      });

      games.push(...gamesData);
      logger.info(`Steam: Scraped ${games.length} games`);

    } catch (error) {
      logger.error('Steam scraping error:', error);
      throw error;
    }

    return games;
  }

  async scrapeGameDetails(slug: string): Promise<any> {
    if (!this.page) throw new Error('Scraper not initialized');

    try {
      await this.page.goto(`https://store.steampowered.com/app/${slug}`, {
        waitUntil: 'networkidle2',
        timeout: 30000,
      });

      const details = await this.page.evaluate(() => {
        const name = document.querySelector('.apphub_AppName')?.textContent?.trim() || '';
        const description = document.querySelector('.game_description_snippet')?.textContent?.trim() || '';
        const releaseDate = document.querySelector('.date')?.textContent?.trim() || '';
        
        // Extrair gêneros
        const genreElements = document.querySelectorAll('.game_area_details_stats_ctn a');
        const genres = Array.from(genreElements).map(el => el.textContent?.trim() || '');

        // Extrair preços
        const priceElement = document.querySelector('.game_purchase_price');
        const originalPriceElement = document.querySelector('.discount_original_price');
        const discountElement = document.querySelector('.discount_pct');

        const currentPrice = priceElement ? 
          parseFloat(priceElement.textContent?.replace(/[^\d,]/g, '').replace(',', '.') || '0') : 0;
        const originalPrice = originalPriceElement ? 
          parseFloat(originalPriceElement.textContent?.replace(/[^\d,]/g, '').replace(',', '.') || '0') : currentPrice;
        const discountPercent = discountElement ? 
          parseInt(discountElement.textContent?.replace(/[^0-9]/g, '') || '0') : 0;

        return {
          name,
          description,
          releaseDate,
          genres,
          currentPrice,
          originalPrice,
          discountPercent,
        };
      });

      return details;

    } catch (error) {
      logger.error(`Steam game details error for ${slug}:`, error);
      throw error;
    }
  }

  async saveToDatabase(games: SteamGame[]): Promise<void> {
    for (const game of games) {
      try {
        // Upsert game
        const gameResult = await db.query(
          `INSERT INTO games (slug, name, image_url)
           VALUES ($1, $2, $3)
           ON CONFLICT (slug) DO UPDATE SET 
             name = EXCLUDED.name,
             image_url = EXCLUDED.image_url,
             updated_at = NOW()
           RETURNING id`,
          [game.slug, game.name, game.imageUrl]
        );

        const gameId = gameResult.rows[0].id;

        // Upsert price
        await db.query(
          `INSERT INTO prices (game_id, store, current_price, original_price, discount_percent, url)
           VALUES ($1, 'steam', $2, $3, $4, $5)
           ON CONFLICT (game_id, store) DO UPDATE SET 
             current_price = EXCLUDED.current_price,
             original_price = EXCLUDED.original_price,
             discount_percent = EXCLUDED.discount_percent,
             url = EXCLUDED.url,
             last_checked = NOW()`,
          [gameId, game.currentPrice, game.originalPrice, game.discountPercent, game.url]
        );

        // Save price history
        await db.query(
          `INSERT INTO price_history (price_id, price)
           SELECT id, $1 FROM prices WHERE game_id = $2 AND store = 'steam'`,
          [game.currentPrice, gameId]
        );

      } catch (error) {
        logger.error(`Error saving game ${game.slug}:`, error);
      }
    }
  }

  async cleanup(): Promise<void> {
    if (this.browser) {
      await this.browser.close();
    }
  }
}
```

```typescript
import { Queue, Worker } from 'bullmq';
import { Redis } from 'ioredis';
import { SteamScraper } from '../scrapers/steam-scraper';

// Criar queue
export const scrapingQueue = new Queue('scraping', { connection });

// Worker para processar jobs
const scrapingWorker = new Worker(
  'scraping',
  async (job) => {
    const { store, type } = job.data;
    
    logger.info(`Processing scraping job: ${store} - ${type}`);

    try {
      if (store === 'steam') {
        const scraper = new SteamScraper();
        await scraper.initialize();
        
        const games = await scraper.scrapeTopDeals();
        await scraper.saveToDatabase(games);
        await scraper.cleanup();
        
        return { success: true, gamesCount: games.length };
        
      } else if (store === 'epic') {
        const scraper = new EpicScraper();
        await scraper.initialize();
        
        const freeGames = await scraper.scrapeFreeGames();
        const allDeals = await scraper.scrapeAllDeals();
        
        await scraper.cleanup();
        
        return { 
          success: true, 
          freeGamesCount: freeGames.length,
          dealsCount: allDeals.length 
        };
      }
    } catch (error) {
      logger.error(`Scraping job failed: ${store} - ${type}`, error);
      throw error;
    }
  },
  { 
    connection,
    concurrency: 3, // Processar 3 jobs simultaneamente
    limiter: {
      max: 10,
      duration: 60000, // Máximo 10 jobs por minuto
    },
  }
);

// Eventos do worker
scrapingWorker.on('completed', (job, result) => {
  logger.info(`Job ${job.id} completed:`, result);
});

});
```

```typescript
// src/middleware/auth.ts

import { Request, Response, NextFunction } from 'express';
import { db } from '../database';
import { logger } from '../utils/logger';

export interface AuthRequest extends Request {
  userId?: number;
  plan?: string;
}

export async function authMiddleware(
  req: AuthRequest,
  res: Response,
  next: NextFunction
): Promise<void> {
  try {
    const authHeader = req.headers.authorization;
    
    if (!authHeader || !authHeader.startsWith('Bearer ')) {
      res.status(401).json({ 
        error: 'Missing or invalid authorization header' 
      });
      return;
    }

    const apiKey = authHeader.split(' ')[1];

    // Buscar usuário pela API key
    const result = await db.query(
      `SELECT id, plan, requests_limit, requests_used 
       FROM api_users 
       WHERE api_key = $1`,
      [apiKey]
    );

    if (result.rows.length === 0) {
      res.status(401).json({ error: 'Invalid API key' });
      return;
    }

    const user = result.rows[0];

    // Verificar rate limit do plano
    if (user.requests_used >= user.requests_limit) {
      res.status(429).json({ 
        error: 'Rate limit exceeded',
        limit: user.requests_limit,
        used: user.requests_used,
        upgrade_url: 'https://api.example.com/pricing',
      });
      return;
    }

    // Adicionar dados do usuário ao request
    req.userId = user.id;
    req.plan = user.plan;

    // Incrementar uso (async, não bloqueia)
    db.query(
      `UPDATE api_users SET requests_used = requests_used + 1 WHERE id = $1`,
      [user.id]
    ).catch(err => logger.error('Error incrementing usage:', err));

    next();

  } catch (error) {
    logger.error('Auth middleware error:', error);
    res.status(500).json({ error: 'Internal server error' });
  }
}
```

```typescript
// src/websocket.ts

import { Server } from 'socket.io';
import { Server as HttpServer } from 'http';
import { redis } from './redis';
import { logger } from './utils/logger';

let io: Server;

export function setupWebSocket(server: HttpServer): void {
  io = new Server(server, {
    cors: {
      origin: '*',
      methods: ['GET', 'POST'],
    },
  });

  io.on('connection', (socket) => {
    logger.info(`Client connected: ${socket.id}`);

    // Autenticar cliente
    socket.on('authenticate', async (apiKey: string) => {
      // Validar API key (simplificado)
      socket.emit('authenticated', { success: true });
    });

    // Entrar em sala de preço de jogo específico
    socket.on('subscribe:price', (gameSlug: string) => {
      socket.join(`price:${gameSlug}`);
      logger.info(`Client ${socket.id} subscribed to price:${gameSlug}`);
    });

    // Entrar em sala de ofertas
    socket.on('subscribe:deals', () => {
      socket.join('deals');
      logger.info(`Client ${socket.id} subscribed to deals`);
    });

    socket.on('disconnect', () => {
      logger.info(`Client disconnected: ${socket.id}`);
    });
  });

  // Escutar mudanças no Redis (pub/sub)
  const subscriber = redis.duplicate();
  subscriber.subscribe('price-changes');
  subscriber.subscribe('new-deals');

  subscriber.on('message', (channel, message) => {
    const data = JSON.parse(message);
    
    if (channel === 'price-changes') {
      io.to(`price:${data.gameSlug}`).emit('price-update', data);
    } else if (channel === 'new-deals') {
      io.to('deals').emit('new-deal', data);
    }
  });

  logger.info('WebSocket server initialized');
}

// Função para publicar mudanças de preço
export async function publishPriceChange(gameSlug: string, data: any): Promise<void> {
  await redis.publish('price-changes', JSON.stringify({ gameSlug, ...data }));
}

// Função para publicar nova oferta
export async function publishNewDeal(data: any): Promise<void> {
  await redis.publish('new-deals', JSON.stringify(data));
}
```
